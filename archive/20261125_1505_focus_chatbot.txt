function chatMatch(msg){
  const low = msg.toLowerCase();
  for(let i=0;i<KB_I18N.keywords.length;i++){
    if(KB_I18N.keywords[i].some(kw=>low.includes(kw))) return KB_I18N.a[currentLang][i];
  }
  return KB_I18N.a[currentLang][KB_I18N.a[currentLang].length-1] || t('chatWelcome');
}
function addMsg(text,cls){
  const div=document.createElement('div'); div.className='cmsg '+cls; div.textContent=text;
  const m=document.getElementById('chat-msgs'); m.appendChild(div); m.scrollTop=9999;
}
let chatOpen=false;
function toggleChat(){
  chatOpen=!chatOpen;
  document.getElementById('chat-box').classList.toggle('open',chatOpen);
  if(chatOpen && document.getElementById('chat-msgs').children.length===0){
    setTimeout(()=>addMsg(t('chatWelcome'),'ai'),300);
  }
}
document.getElementById('chat-input').addEventListener('keydown', e=>{
  if(e.key==='Enter'){
    const v = e.target.value.trim(); if(!v) return;
    addMsg(v,'user'); e.target.value='';
    setTimeout(()=>addMsg(chatMatch(v),'ai'), 300);
  }
});
